cmake_minimum_required(VERSION 3.16)
project(drift CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()
add_compile_options(-Wall -Wextra)

# SDL3 through pkg-config (its exported CMake config is broken here)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL3 REQUIRED IMPORTED_TARGET sdl3)
add_library(SDL3::SDL3 ALIAS PkgConfig::SDL3)
find_package(Threads REQUIRED)

file(GLOB CORE_SOURCES src/*.cpp)
list(REMOVE_ITEM CORE_SOURCES ${CMAKE_SOURCE_DIR}/src/main.cpp)
add_library(core STATIC ${CORE_SOURCES})
target_include_directories(core PUBLIC src)
target_link_libraries(core PUBLIC Threads::Threads)

add_executable(drift src/main.cpp)
target_link_libraries(drift PRIVATE core SDL3::SDL3)

enable_testing()
add_executable(drift_tests tests/tests.cpp)
target_link_libraries(drift_tests PRIVATE core)
add_test(NAME drift_tests COMMAND drift_tests ${CMAKE_SOURCE_DIR})
set_tests_properties(drift_tests PROPERTIES TIMEOUT 3600)
